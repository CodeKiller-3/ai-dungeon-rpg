(function () {
  'use strict';
  var q = {};
  location.search.replace(/^\?/, '').split('&').forEach(function (kv) {
    var p = kv.split('='); if (p[0]) q[decodeURIComponent(p[0])] = decodeURIComponent(p[1] || '');
  });
  if (q.w) {
    Object.defineProperty(window, 'innerWidth', { value: +q.w, configurable: true });
    Object.defineProperty(window, 'innerHeight', { value: +q.h || 640, configurable: true });
  }
  var M = Game.Core.Main, P = Game.Data.Player, B = Game.Systems.Battle, Map = Game.Systems.Map;
  var R = Game.Core.Render;

  /* headless 不稳帧率：把 FPS 段从底部提示里去掉（与交付图口径一致） */
  (function hideHeadlessFps() {
    var proto = window.CanvasRenderingContext2D && window.CanvasRenderingContext2D.prototype;
    if (!proto || proto.__d88fps) return;
    var orig = proto.fillText;
    proto.fillText = function (text, x, y, maxWidth) {
      if (typeof text === 'string' && text.indexOf('FPS=') === 0) {
        text = text.replace(/^FPS=[^|]*\|\s*/, '');
      }
      if (arguments.length > 3) return orig.call(this, text, x, y, maxWidth);
      return orig.call(this, text, x, y);
    };
    proto.__d88fps = true;
  })();

  var spriteLoad = Game.Core.Sprites.load;
  Game.Core.Sprites.load = function () { return spriteLoad.call(this, '../../../assets/sprites/'); };
  var getCalls = 0, getTile = 0;
  var spriteGet = Game.Core.Sprites.get;
  Game.Core.Sprites.get = function (n) {
    getCalls++;
    if (typeof n === 'string' && n.indexOf('floor_') >= 0) getTile++;
    return spriteGet.call(this, n);
  };

  function boot() {
    try {
      M.init({ menu: false });
      var f = Math.max(1, Math.min(5, +(q.f || 1)));
      Map.setFloor(f);
      P.floor = f;
      P.level = 5;
      P.exp = 24;
      P.gold = 268;
      P.hp = 100;
      P.maxHp = 140;
      P.mp = 30;
      P.maxMp = 30;
      P.atk = 14;
      P.def = 6;
      B.start(q.m || 'slime');
      M.switchScene('battle');
      R.resize();
      R._animNow = function () { return 1234; };
      window.__d88F = f;
      window.__d88Booted = true;
    } catch (e) {
      document.getElementById('err').textContent = String(e && (e.stack || e.message) || e);
      document.title = 'ERR';

  function spritesReady(f) {
    var S = Game.Core.Sprites;
    if (!S.get('t' + f + '_floor_1') || !S.get('knight_m_idle_anim_f0')) return false;
    var mid = q.m || 'slime';

  })();
})();
